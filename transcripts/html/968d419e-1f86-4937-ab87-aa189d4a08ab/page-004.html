import { useCallback, useState, useSyncExternalStore } from 'react'

const WIDE_QUERY = '(min-width: 1280px)'

function subscribeWide(cb: () => void): () => void {
  if (typeof window === 'undefined' || typeof window.matchMedia !== 'function') return () => undefined
  const mql = window.matchMedia(WIDE_QUERY)
  mql.addEventListener('change', cb)
  return () => mql.removeEventListener('change', cb)
}

function isWide(): boolean {
  return typeof window !== 'undefined' && typeof window.matchMedia === 'function'
    ? window.matchMedia(WIDE_QUERY).matches
    : true
}

/** True at ≥1280 px, where the workspace is a side column rather than an overlay sheet. */
export function useIsWide(): boolean {
  return useSyncExternalStore(subscribeWide, isWide, () => true)
}

/**
 * Workspace panel open state: open by default on wide screens (a side column), closed by default
 * on narrow ones (where it is an overlay sheet the reader must ask for). Follows the breakpoint on
 * resize so rotating a tablet does not leave a sheet covering the transcript: the user's choice is
 * kept per breakpoint, and a breakpoint change falls back to that breakpoint's default.
 */
export function useWorkspaceOpen(): [boolean, (open: boolean | ((v: boolean) => boolean)) => void] {
  const wide = useIsWide()
  const [choice, setChoice] = useState<{ wide: boolean; open: boolean } | null>(null)
  const open = choice && choice.wide === wide ? choice.open : wide
  const setOpen = useCallback(
    (next: boolean | ((v: boolean) => boolean)) =>
      setChoice((prev) => {
        const cur = prev && prev.wide === wide ? prev.open : wide
        return { wide, open: typeof next === 'function' ? next(cur) : next }
      }),
    [wide],
  )
  return [open, setOpen]
}

What was wrong. The resizable split is remembered in the browser, and that memory included a collapsed workspace. So once you collapsed it, or dragged it to zero, every later conversation or replay opened with the panel hidden.

What changed. A remembered split is now reused only if the workspace was actually visible in it; a collapsed or near-zero split is ignored, so each new conversation or replay opens with the panel on screen at the normal width. You can still collapse it for the page you're on. I also stopped the panel's own resize callback from firing during mount, since a transient collapsed measurement could otherwise flip the toggle off.

Verified on production. On the lost-ack replay I collapsed the panel to zero width, navigated to the gauntlet replay, and the panel was back at 491 px with the Workspace toggle active. Conversation pages use the same layout and remount per conversation, so they get the same behaviour. Below 1280 px the workspace remains an overlay sheet that starts closed, since opening it there would cover the transcript.
